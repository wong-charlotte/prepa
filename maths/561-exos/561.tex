\begin{align*}
    f:x \mapsto \sum_{n=0}^{+\infty}\sin(q^{n}x)
\end{align*}
\(\forall x \in \mathbb{R}, \forall n \in \mathbb{N}\)
\begin{align*}
  |\sin(q^{n}x)| \leq |x q^{n}| \indent \text{t.g. série convergente}
\end{align*}
Donc f(x) est définie sur \(\mathbb{R}\)
